CREATE TABLE utilisateurs (
    id INT AUTO_INCREMENT PRIMARY KEY,    -- ID unique pour chaque utilisateur
    nom VARCHAR(100) NOT NULL,            -- Nom de l'utilisateur
    prenom VARCHAR(100) NOT NULL,         -- Prénom de l'utilisateur
    mot_de_passe VARCHAR(255) NOT NULL,   -- Mot de passe de l'utilisateur
    date_inscription DATETIME DEFAULT CURRENT_TIMESTAMP, -- Date d'inscription
    date_naissance DATE,                  -- Date de naissance de l'utilisateur
    telephone VARCHAR(20),                -- Numéro de téléphone
    langue VARCHAR(50),                   -- Langue de l'utilisateur
    email VARCHAR(255) UNIQUE NOT NULL,   -- Adresse e-mail (unique)
    adresse VARCHAR(255)                  -- Adresse de l'utilisateur
);

CREATE TABLE utilisateurs_reservations (
    id_utilisateur INT,
    id_reservation INT,
    nombreJoueur INT NOT NULL, 
    statut VARCHAR(50) NOT NULL DEFAULT 'Réservé',
    PRIMARY KEY (id_utilisateur, id_reservation),
    FOREIGN KEY (id_utilisateur) REFERENCES utilisateurs(id) ON DELETE CASCADE,
    FOREIGN KEY (id_reservation) REFERENCES reservations(id_reservation) ON DELETE CASCADE
);

CREATE TABLE items (
    id_item INT AUTO_INCREMENT PRIMARY KEY,
    id_utilisateur INT,
    id_reservation INT,  
    nombreJoueur INT NOT NULL, 
    FOREIGN KEY (id_utilisateur) REFERENCES utilisateurs(id) ON DELETE CASCADE,
    FOREIGN KEY (id_reservation) REFERENCES reservations(id_reservation) ON DELETE CASCADE
);

CREATE TABLE reservations_installations (
    id_reservation INT NOT NULL,
    id_installation INT NOT NULL,
    capacite INT NOT NULL,
    PRIMARY KEY (id_reservation, id_installation),
    FOREIGN KEY (id_reservation) REFERENCES reservations(id_reservation) ON DELETE CASCADE,
    FOREIGN KEY (id_installation) REFERENCES installations(id_installation) ON DELETE CASCADE
);
